% Definitions required by Pandoc-generated LaTeX (latex/generated/*.tex).
% Load after graphicx. Do not remove.

\makeatletter

% Resolve \input and \includegraphics paths relative to the project root.
\def\input@path{{../}}
\graphicspath{{../}}

\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

% \pandocbounded (Pandoc >= 3.2): scale an image down to fit the line width.
\newsavebox\pandoc@box
\providecommand*\pandocbounded[1]{%
  \sbox\pandoc@box{#1}%
  \Gscale@div\@tempa{\textheight}{\dimexpr\ht\pandoc@box+\dp\pandoc@box\relax}%
  \Gscale@div\@tempb{\linewidth}{\wd\pandoc@box}%
  \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi
  \ifdim\@tempa\p@<\p@\scalebox{\@tempa}{\usebox\pandoc@box}%
  \else\usebox{\pandoc@box}%
  \fi}

\makeatother
